-- ~/.config/wezterm/wezterm.lua
-- Configuração principal do WezTerm com integração de IA + transparência macOS

local wezterm = require 'wezterm'
local act = wezterm.action
local config = wezterm.config_builder()

-- Carregar módulos customizados
local theme = require 'lua.theme'
local ai_bindings = require 'lua.ai-bindings'
local ai_pane = require 'lua.ai-pane'
local keybindings = require 'lua.keybindings'
local transparency = require 'lua.transparency'
local blur = require 'lua.blur'

-- ============================================
-- CONFIGURAÇÕES BÁSICAS
-- ============================================

config.font_size = 12.0
config.font = wezterm.font_with_fallback {
  'JetBrains Mono',
  'Fira Code',
  'Cascadia Code',
  'Hack',
  'Menlo',
  'monospace',
}

-- Habilitar ligatures de fonte
config.harfbuzz_features = { 'calt', 'liga', 'dlig' }

-- ============================================
-- TRANSPARÊNCIA + BLUR ESTILO MACOS
-- ============================================
-- Janela em foco: 85% opaco (permite blur visível)
-- Janela sem foco: ajustada via eventos (75% estilo macOS)
-- Blur de fundo é aplicado pelo compositor (KDE/KWin) automaticamente

-- Opacidade do background da janela (permite ver o wallpaper com blur)
config.window_background_opacity = 0.85  -- 85% opaco, 15% transparente para blur

-- Habilitar blur no macOS
config.macos_window_background_blur = 30

-- Habilitar Wayland para blur nativo via compositor
config.enable_wayland = true

-- ============================================
-- TEMA E CORES
-- ============================================

theme.apply(config)

-- ============================================
-- SPLITS E NAVEGAÇÃO
-- ============================================

config.keys = {
  -- Splits nativos do WezTerm
  { key = 'd', mods = 'CTRL|SHIFT', action = act.SplitVertical { domain = 'CurrentPaneDomain' } },
  { key = 'e', mods = 'CTRL|SHIFT', action = act.SplitHorizontal { domain = 'CurrentPaneDomain' } },

  -- Navegação entre panes (Alt + Setas)
  { key = 'LeftArrow',  mods = 'ALT', action = act.ActivatePaneDirection 'Left' },
  { key = 'RightArrow', mods = 'ALT', action = act.ActivatePaneDirection 'Right' },
  { key = 'UpArrow',    mods = 'ALT', action = act.ActivatePaneDirection 'Up' },
  { key = 'DownArrow',  mods = 'ALT', action = act.ActivatePaneDirection 'Down' },

  -- Redimensionar panes (Alt+Shift + Setas)
  { key = 'LeftArrow',  mods = 'ALT|SHIFT', action = act.AdjustPaneSize { 'Left', 5 } },
  { key = 'RightArrow', mods = 'ALT|SHIFT', action = act.AdjustPaneSize { 'Right', 5 } },
  { key = 'UpArrow',    mods = 'ALT|SHIFT', action = act.AdjustPaneSize { 'Up', 5 } },
  { key = 'DownArrow',  mods = 'ALT|SHIFT', action = act.AdjustPaneSize { 'Down', 5 } },

  -- Fechar pane
  { key = 'w', mods = 'CTRL|SHIFT', action = act.CloseCurrentPane { confirm = false } },

  -- Tabs
  { key = 't', mods = 'CTRL|SHIFT', action = act.SpawnTab 'CurrentPaneDomain' },
  { key = 'Tab', mods = 'CTRL', action = act.ActivateTabRelative(1) },
  { key = 'Tab', mods = 'CTRL|SHIFT', action = act.ActivateTabRelative(-1) },
  { key = '1', mods = 'CTRL', action = act.ActivateTab(0) },
  { key = '2', mods = 'CTRL', action = act.ActivateTab(1) },
  { key = '3', mods = 'CTRL', action = act.ActivateTab(2) },
  { key = '4', mods = 'CTRL', action = act.ActivateTab(3) },
  { key = '5', mods = 'CTRL', action = act.ActivateTab(4) },
  { key = '6', mods = 'CTRL', action = act.ActivateTab(5) },
  { key = '7', mods = 'CTRL', action = act.ActivateTab(6) },
  { key = '8', mods = 'CTRL', action = act.ActivateTab(7) },
  { key = '9', mods = 'CTRL', action = act.ActivateTab(-1) },

  -- JANELA
  { key = 'Enter', mods = 'ALT', action = act.ToggleFullScreen },

  -- ZOOM DE PANE
  { key = 'z', mods = 'CTRL|SHIFT', action = act.TogglePaneZoomState },
  { key = 'r', mods = 'CTRL|SHIFT', action = act.ClearScrollback 'ScrollbackAndViewport' },

  -- FONTE
  { key = '+', mods = 'CTRL', action = act.IncreaseFontSize },
  { key = '=', mods = 'CTRL', action = act.IncreaseFontSize },
  { key = '-', mods = 'CTRL', action = act.DecreaseFontSize },
  { key = '0', mods = 'CTRL', action = act.ResetFontSize },

  -- LISTA DE ATALHOS (Ctrl+Shift+K)
  { key = 'k', mods = 'CTRL|SHIFT', action = wezterm.action_callback(function(window, pane)
      keybindings.show_in_tab(window, pane)
    end)
  },

  -- COMMAND PALETTE
  { key = '?', mods = 'CTRL|SHIFT', action = act.ActivateCommandPalette },

  -- SCROLL / COPY MODE
  { key = 'Space', mods = 'CTRL|SHIFT', action = act.ActivateCopyMode },

  -- BUSCA
  { key = 'f', mods = 'CTRL|SHIFT', action = act.Search { CaseInSensitiveString = '' } },
}

-- Adicionar keybindings de IA
ai_bindings.apply(config, act)
ai_pane.setup_ai_keys(config, act)
ai_pane.register_events()

-- Aplicar transparência (registra eventos de foco)
transparency.apply(config)

-- ============================================
-- BARRA DE STATUS
-- ============================================

wezterm.on('update-right-status', function(window, pane)
  local date = wezterm.strftime '%H:%M:%S'
  local cwd = pane:get_current_working_dir()

  local cwd_text = ''
  if cwd and cwd.file_path then
    local path = cwd.file_path
    local basename = path:match('([^/]+)$') or path
    cwd_text = ' ' .. basename .. ' '
  end

  window:set_right_status(wezterm.format {
    { Foreground = { Color = '#7aa2f7' } },
    { Text = cwd_text },
    { Foreground = { Color = '#565f89' } },
    { Text = ' ' .. date .. ' ' },
  })
end)

-- ============================================
-- CONFIGURAÇÕES DE JANELA
-- ============================================

config.window_padding = {
  left = 8,
  right = 8,
  top = 8,
  bottom = 8,
}

config.window_decorations = 'RESIZE'
config.use_fancy_tab_bar = false
config.show_tab_index_in_tab_bar = true
config.tab_bar_at_bottom = false

-- Habilitar shell integration
config.enable_kitty_keyboard = true
config.enable_wayland = true

return config
